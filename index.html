SistemaEducativoIA: """Classe controladora principal do sistema.""" def __init__(self, nome_aluno: str, porta_robot: Optional[str] = None): self.aluno = Aluno(nome_aluno) self.motor = MotorAdaptativo() self.robot = ControladorRobot(porta_robot) self.banco = GestorBancoDados() def guardar_progresso(self, ficheiro: str = "progresso_aluno.json"): dados = { "aluno": self.aluno.nome,
